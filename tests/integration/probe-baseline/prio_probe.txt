########## getpriority: which and who ##########
getpriority(PRIO_PROCESS, 0)                         -> 20 (errno clear)
getpriority(PRIO_PGRP, 0)                            -> 20 (errno clear)
getpriority(PRIO_USER, 0)                            -> 20 (errno clear)
getpriority(PRIO_PROCESS, self)                      -> 20 (errno clear)
getpriority(PRIO_PROCESS, 999999)                    -> -1 (errno ESRCH)
getpriority(PRIO_PROCESS, -1)                        -> -1 (errno ESRCH)
getpriority(3, 0) -- no such `which`                 -> -1 (errno EINVAL)
getpriority(-1, 0)                                   -> -1 (errno EINVAL)

########## setpriority: the range and the clamp ##########
setpriority(PRIO_PROCESS, 0, 0)                      -> 0 then 20 (errno clear)
setpriority(PRIO_PROCESS, 0, 5)                      -> 0 then 15 (errno clear)
setpriority(PRIO_PROCESS, 0, 19)                     -> 0 then 1 (errno clear)
setpriority(PRIO_PROCESS, 0, 20)                     -> 0 then 1 (errno clear)
setpriority(PRIO_PROCESS, 0, 100)                    -> 0 then 1 (errno clear)
setpriority(PRIO_PROCESS, 0, -1)                     -> EACCES then 1 (errno clear)
setpriority(PRIO_PROCESS, 0, -20)                    -> EACCES then 1 (errno clear)
setpriority(PRIO_PROCESS, 0, -21)                    -> EACCES then 1 (errno clear)
setpriority(PRIO_PROCESS, 0, -100)                   -> EACCES then 1 (errno clear)
setpriority(PRIO_PROCESS, 0, -2147483648)            -> EACCES then 1 (errno clear)
setpriority(3, 0, 0) -- no such `which`              -> EINVAL
setpriority(PRIO_PROCESS, 999999, 0)                 -> ESRCH
